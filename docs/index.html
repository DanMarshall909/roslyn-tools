Three ways to workcompiler-backed
# CLI: inspect a type
rt inspect describe BookingService \
  --solution MyApp.slnx --trust-repository

# MCP: an agent follows callers
relationships {
  "symbol": "BookingService.ReserveAsync",
  "mode": "callers",
  "solution": "MyApp.slnx"
}

# CI: enforce your .rt/ci.json policy
rt ci check --solution MyApp.slnx \
  --json --trust-repository
